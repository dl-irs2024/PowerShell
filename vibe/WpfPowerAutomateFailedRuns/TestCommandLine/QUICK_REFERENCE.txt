  $button.Add_Click({
      Invoke-TestStep -Step 'Sample Environments' -Action {
          $result = Test-AdminEnvironments
          if ($result.Success) {
              $testSampleGrid.ItemsSource = $result.Environments
              return "$($result.Count) environment(s)"
